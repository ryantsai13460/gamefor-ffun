2. server.mjs This file supplies real web-search and YouTube keyword results. It is not a general-purpose proxy. // Node.js 22+. // API keys stay on the server, never in index.html. import http from 'node:http'; import { readFile } from 'node:fs/promises'; const PORT = Number(process.env.PORT || 3000); const HOST = process.env.HOST || '127.0.0.1'; const ROOT = new URL('./', import.meta.url); const allowedHosts = new Set([ `127.0.0.1:${PORT}`, `localhost:${PORT}` ]); if (process.env.PUBLIC_ORIGIN) { allowedHosts.add(new URL(process.env.PUBLIC_ORIGIN).host); } const keys = { web: process.env.BRAVE_API_KEY, youtube: process.env.YOUTUBE_API_KEY }; const visitors = new Map(); const clean = value => String(value || '').replace(/<[^>]*>/g, '').slice(0, 1800); const httpsURL = value => { try { const u = new URL(value); return u.protocol === 'https:' && !u.username && !u.password ? u.href : ''; } catch { return ''; } }; const headers = { 'X-Content-Type-Options': 'nosniff', 'Referrer-Policy': 'strict-origin-when-cross-origin', 'X-Frame-Options': 'SAMEORIGIN', 'Cache-Control': 'no-store' }; function send(res, status, data) { res.writeHead(status, { ...headers, 'Content-Type': 'application/json; charset=utf-8' }); res.end(JSON.stringify(data)); } function limited(ip) { const now = Date.now(); for (const [key, v] of visitors) { if (v.until < now) visitors.delete(key); } if (!visitors.has(ip)) { if (visitors.size >= 2000) return true; visitors.set(ip, { count: 0, until: now + 60000 }); } return ++visitors.get(ip).count > 12; } async function providerJSON(url, extraHeaders = {}) { const response = await fetch(url, { headers: { Accept: 'application/json', ...extraHeaders }, redirect: 'error', signal: AbortSignal.timeout(12000) }); if (!response.ok) { const error = new Error( response.status === 429 ? 'Search quota reached. Try again later.' : 'Search provider rejected the request. Check the server API key and quota.' ); error.status = 502; throw error; } const reader = response.body.getReader(); let size = 0; const chunks = []; for (;;) { const { done, value } = await reader.read(); if (done) break; size += value.byteLength; if (size > 1000000) { await reader.cancel(); throw new Error('Search response was too large.'); } chunks.push(Buffer.from(value)); } return JSON.parse(Buffer.concat(chunks).toString('utf8')); } const server = http.createServer(async (req, res) => { try { if ((req.url || '').length > 4096) { return send(res, 414, { error: 'Request is too long.' }); } if (!allowedHosts.has(req.headers.host)) { return send(res, 403, { error: 'Unexpected host name.' }); } const base = new URL('http://' + req.headers.host); const url = new URL(req.url, base); if (req.method !== 'GET' && req.method !== 'HEAD') { return send(res, 405, { error: 'GET requests only.' }); } if (url.pathname.startsWith('/api/')) { const origin = req.headers.origin; let sameOrigin = !origin; try { if (origin) sameOrigin = new URL(origin).host === base.host; } catch {} if ( req.headers['sec-fetch-site'] === 'cross-site' || !sameOrigin ) { return send(res, 403, { error: 'Open the app on this server before searching.' }); } if (url.pathname === '/api/status') { return send(res, 200, { web: Boolean(keys.web), youtube: Boolean(keys.youtube) }); } const kind = url.pathname === '/api/web' ? 'web' : url.pathname === '/api/youtube' ? 'youtube' : null; if (!kind) { return send(res, 404, { error: 'Unknown API route.' }); } if (!keys[kind]) { return send(res, 503, { error: `The server needs ${ kind === 'web' ? 'BRAVE_API_KEY' : 'YOUTUBE_API_KEY' }. Games and pasted video links do not need this key.` }); } const q = (url.searchParams.get('q') || '').trim(); if (!q || q.length > 200 || /[\x00-\x1f]/.test(q)) { return send(res, 400, { error: 'Enter 1–200 characters.' }); } if (limited(req.socket.remoteAddress)) { return send(res, 429, { error: 'Please wait a minute before searching again.' }); } // Fixed official endpoints only. // No arbitrary URL fetch or website relay. if (kind === 'web') { const endpoint = new URL( 'https://api.search.brave.com/res/v1/web/search' ); endpoint.search = new URLSearchParams({ q, count: '10', safesearch: 'strict', text_decorations: 'false' }); const data = await providerJSON(endpoint, { 'X-Subscription-Token': keys.web }); return send(res, 200, { items: (data.web?.results || []) .slice(0, 10) .map(x => ({ title: clean(x.title), description: clean(x.description), url: httpsURL(x.url) })) .filter(x => x.url) }); } const endpoint = new URL( 'https://www.googleapis.com/youtube/v3/search' ); endpoint.search = new URLSearchParams({ key: keys.youtube, part: 'snippet', q, type: 'video', maxResults: '10', safeSearch: 'strict', videoEmbeddable: 'true', videoSyndicated: 'true' }); const data = await providerJSON(endpoint); return send(res, 200, { items: (data.items || []) .filter(x => /^[\w-]{11}$/.test(x.id?.videoId || '')) .map(x => ({ id: x.id.videoId, title: clean(x.snippet?.title), description: clean(x.snippet?.channelTitle), image: `https://i.ytimg.com/vi/${x.id.videoId}/mqdefault.jpg` })) }); } if (url.pathname === '/' || url.pathname === '/index.html') { const html = await readFile(new URL('index.html', ROOT)); res.writeHead(200, { ...headers, 'Content-Type': 'text/html; charset=utf-8' }); return res.end(req.method === 'HEAD' ? undefined : html); } if (url.pathname === '/favicon.ico') { res.writeHead(204); return res.end(); } send(res, 404, { error: 'Not found.' }); } catch (error) { send(res, error.status || 502, { error: error.name === 'TimeoutError' ? 'Search timed out. Try again.' : error.message || 'Request failed.' }); } }); server.requestTimeout = 15000; server.headersTimeout = 10000; server.listen(PORT, HOST, () => { console.log(`gamefor ffun: http://${HOST}:${PORT}`); });
